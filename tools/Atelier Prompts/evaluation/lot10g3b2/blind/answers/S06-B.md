<script>
const quiz=document.getElementById("quiz");
const sections=[...quiz.querySelectorAll("section")];
document.getElementById("check").addEventListener("click",()=>{
 let score=0;
 sections.forEach((section,index)=>{
  const choice=section.querySelector(`input[name="q${index+1}"]:checked`);
  const feedback=section.querySelector(".feedback");
  if(choice&&choice.value===section.dataset.answer){
   score++;
   feedback.textContent="Correct. "+section.dataset.text;
   feedback.className="feedback correct";
  }else{
   feedback.textContent="Réponse incorrecte. "+section.dataset.text;
   feedback.className="feedback incorrect";
  }
 });
 document.getElementById("score").textContent=`Score : ${score} sur ${sections.length}`;
});
document.getElementById("reset").addEventListener("click",()=>{
 sections.forEach(section=>{
  const feedback=section.querySelector(".feedback");
  feedback.textContent="";
  feedback.className="feedback";
 });
 document.getElementById("score").textContent="";
});
</script>
