```js
function uniqueSorted(numbers) {
  return [...new Set(numbers)].sort((a, b) => a - b);
}
```

### Exemples

```js
console.log(uniqueSorted([4, 2, 2, 1, 4, 3]));
// [1, 2, 3, 4]

console.log(uniqueSorted([10, -2, 5, 10, -2, 0]));
// [-2, 0, 5, 10]

console.log(uniqueSorted([3.5, 1.2, 3.5, 2.1]));
// [1.2, 2.1, 3.5]
```
